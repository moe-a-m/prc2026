```sql
WITH b AS (SELECT airport, strftime(month,'%Y-%m') mon, f_no_nm_match um, second(t_mvt)=0 mvt_s0,
   epoch(t_mvt)-epoch(t_sched) > 10800 late3h, epoch(t_mvt)-epoch(t_sched) > 21600 late6h, actype_mvt IS NULL actype_miss,
   t_aobt3 IS NULL aobt3_miss, (CASE WHEN flight_mvt IS NULL THEN 'null'
  WHEN regexp_matches(flight_mvt, '^[A-Z]{3}[0-9][0-9A-Z]{0,4}$') THEN 'icao3+num'
  WHEN regexp_matches(flight_mvt, '^[A-Z]{4,5}[0-9][0-9A-Z]*$') AND substr(flight_mvt,4,1)=substr(flight_mvt,3,1) THEN 'mangled_repeat'
  WHEN regexp_matches(flight_mvt, '^[A-Z]{4,5}[0-9][0-9A-Z]*$') AND substr(flight_mvt,4,2)=substr(flight_mvt,2,2) THEN 'mangled_repeat'
  WHEN regexp_matches(flight_mvt, '^[A-Z]{4,5}[0-9][0-9A-Z]*$') THEN 'letters4+num'
  WHEN regexp_matches(flight_mvt, '^[A-Z0-9]{2}[0-9]+[A-Z]?$') THEN 'iata2+num'
  WHEN regexp_matches(flight_mvt, '^[A-Z]+$') THEN 'letters_only(reg?)'
  ELSE 'other' END)='mangled_repeat' mangled, flight_mvt IS NULL fn_null,
   f_no_nm_match AND epoch(t_mvt)-epoch(t_sched) > 10800 um_late3h
  FROM clean_dep),
p AS (SELECT * FROM b WHERE mon IN ('2025-01','2025-07','2026-01','2026-07')
      UNION ALL SELECT * REPLACE ('2025-all' AS mon) FROM b WHERE mon LIKE '2025%')
SELECT airport, mon, count(*) n, round(100*avg(um::INT),2) pct_unmatched, round(100*avg(mvt_s0::INT),2) pct_mvt_sec0,
  round(100*avg(late3h::INT),2) pct_late3h, round(100*avg(um_late3h::INT),3) pct_unmatched_late3h, round(100*avg(late6h::INT),3) pct_late6h,
  round(100*avg(actype_miss::INT),2) pct_actype_missing, round(100*avg(aobt3_miss::INT),2) pct_aobt3_missing,
  round(100*avg(mangled::INT),2) pct_mangled, round(100*avg(fn_null::INT),3) pct_flightnum_null
FROM p GROUP BY ALL ORDER BY airport, mon
```
