Where does the monthly ~60 s step in (MVT-AOBT_3)-y come from? Per airport-month medians: DEP proxy error, DEP MVT-AOBT_3, DEP y, ARR MVT-ARVT_3 (NM landing vs airport landing), ARR taxi-in; mean of epoch%60 of MVT / AOBT_3 / ARVT_3 (30 = uniform seconds, 0 = whole minutes). NM-matched rows only; 2025 and 2026.

```sql
WITH dep AS (SELECT airport, month, ds, median(epoch(t_mvt)-epoch(t_aobt3)) FILTER (abs(epoch(t_mvt)-epoch(t_aobt3))<3600) dep_mvt_aobt3,
      median(epoch(t_mvt)-epoch(t_aobt3)-y) FILTER (abs(epoch(t_mvt)-epoch(t_aobt3)-y)<600) dep_proxy_err,
      median(y) FILTER (y BETWEEN 0 AND 3600) dep_y,
      avg(epoch(t_mvt)%60) dep_mvt_mean_sec, avg(epoch(t_aobt3)%60) dep_aobt3_mean_sec
    FROM stg_mvt WHERE phase='DEP' AND flight_id IS NOT NULL AND t_aobt3 IS NOT NULL GROUP BY ALL),
arr AS (SELECT airport, month, ds, median(epoch(t_mvt)-epoch(t_arvt3)) FILTER (abs(epoch(t_mvt)-epoch(t_arvt3))<1800) arr_mvt_arvt3,
      median(epoch(t_block)-epoch(t_mvt)) FILTER (epoch(t_block)-epoch(t_mvt) BETWEEN 0 AND 3600) arr_taxi_in,
      avg(epoch(t_arvt3)%60) arr_arvt3_mean_sec
    FROM stg_mvt WHERE phase='ARR' AND flight_id IS NOT NULL AND t_arvt3 IS NOT NULL GROUP BY ALL)
SELECT dep.airport, strftime(dep.month,'%Y-%m') mon, dep_proxy_err, dep_mvt_aobt3, dep_y, arr_mvt_arvt3, arr_taxi_in,
  round(dep_mvt_mean_sec,1) dep_mvt_mean_sec, round(dep_aobt3_mean_sec,1) dep_aobt3_mean_sec, round(arr_arvt3_mean_sec,1) arr_arvt3_mean_sec
FROM dep JOIN arr USING (airport, month, ds) WHERE dep.airport IN ('EGLL','LIRF','EDDM','LEMD','LEBL','LFPG','LTFM') ORDER BY 1, 2
```
